🔍 VitaInspire Backup Diagnostic Tool

📋 Instructions:

  1. Open your VitaInspire app in another browser tab
  2. Open browser DevTools (F12) in the VitaInspire tab
  3. Copy the script below and paste it into the Console
  4. Come back here and click "Run All Tests"

Step 1: Copy This Script to Your App Console

// VitaInspire Diagnostic Script
window.vitaInspireDiag = {
    env: {
        supabaseUrl: process.env.EXPO_PUBLIC_SUPABASE_URL,
        supabaseKey: process.env.EXPO_PUBLIC_SUPABASE_ANON_KEY,
        backupUrl: process.env.EXPO_PUBLIC_BACKUP_URL,
    },
    testConnection: async function() {
        if (!this.env.supabaseUrl || !this.env.supabaseKey) {
            return { ok: false, error: 'Environment variables not set' };
        }
        try {
            const response = await fetch(`${this.env.supabaseUrl}/rest/v1/fields?select=count`, {
                method: 'GET',
                headers: {
                    'apikey': this.env.supabaseKey,
                    'Authorization': `Bearer ${this.env.supabaseKey}`,
                    'Content-Type': 'application/json'
                }
            });
            return { ok: response.ok, status: response.status };
        } catch (error) {
            return { ok: false, error: error.message };
        }
    },
    testInsert: async function() {
        if (!this.env.supabaseUrl || !this.env.supabaseKey) {
            return { ok: false, error: 'Environment variables not set' };
        }
        try {
            const testData = {
                code: `DIAG-${Date.now()}`,
                label: 'Diagnostic Test',
                location_code: 'DIAG',
                state: 'Test',
                district: 'Test',
                latitude: 12.9716,
                longitude: 77.5946,
                created_at: new Date().toISOString()
            };
            const response = await fetch(`${this.env.supabaseUrl}/rest/v1/fields`, {
                method: 'POST',
                headers: {
                    'apikey': this.env.supabaseKey,
                    'Authorization': `Bearer ${this.env.supabaseKey}`,
                    'Content-Type': 'application/json',
                    'Prefer': 'return=minimal'
                },
                body: JSON.stringify(testData)
            });
            if (response.ok) {
                return { ok: true, fieldCode: testData.code };
            } else {
                const errorText = await response.text();
                return { ok: false, error: errorText, status: response.status };
            }
        } catch (error) {
            return { ok: false, error: error.message };
        }
    }
};
console.log('✅ Diagnostic script loaded. Run window.vitaInspireDiag.testConnection() to test.');
